// PHASE 1: PRE-INSTANTIATE EXACT CLASS CONTAINERS BEFORE STUDENT ALLOCATION function instantiateFixedClassContainers(manualLocks, selectiveConstraints) { const schedule = {}; LINES.forEach(l => schedule[l] = {}); Object.keys(manualLocks).forEach(subj => { const totalSections = manualLocks[subj]; if (totalSections <= 0) return; // Skip disabled/miscoded subjects // Determine target lines for the locked number of sections let assignedLines = []; if (selectiveConstraints[subj]?.lock) { assignedLines = [selectiveConstraints[subj].lock]; } else { // Distribute totalSections across available lines (e.g. 1 section -> Line 1) assignedLines = LINES.slice(0, totalSections); } assignedLines.forEach(line => { schedule[line][subj] = [[]]; // Instantiate closed container }); }); return schedule; } // PHASE 2: STRICT ENROLMENT INTO PRE-EXISTING CONTAINERS ONLY function enrollStudentsClosedSet(students, schedule, bounds) { const unserved = []; students.forEach(st => { st.subjects.forEach(subj => { if (manualLocks[subj] === 0) return; // Find lines that ALREADY contain an instantiated container for this subject const eligibleLines = LINES.filter(line => schedule[line][subj] && !studentSlots[st.id].has(line) ); if (eligibleLines.length === 0) { unserved.push({ id: st.id, subj, reason: 'Line conflict on pre-allocated class line' }); return; } // Enroll in existing section (Optimal -> Soft Overflow to Max) const targetLine = eligibleLines[0]; const targetSection = schedule[targetLine][subj][0]; if (targetSection.length < bounds.max) { targetSection.push(st.id); studentSlots[st.id].add(targetLine); } else { unserved.push({ id: st.id, subj, reason: 'Pre-allocated class at maximum capacity' }); } }); }); return unserved; }
